cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  project(MarkSixStatisticsTests LANGUAGES C CXX)
  enable_testing()
endif()
set(SQLITE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../deps/sqlite-amalgamation-3530400")
if(NOT EXISTS "${SQLITE_DIR}/sqlite3.c")
  message(FATAL_ERROR "Pinned SQLite missing; use scripts/fetch-sqlite.ps1 -Fetch")
endif()
file(SHA256 "${SQLITE_DIR}/sqlite3.c" SQLITE_C_HASH)
file(SHA256 "${SQLITE_DIR}/sqlite3.h" SQLITE_H_HASH)
if(NOT SQLITE_C_HASH STREQUAL "b1dd5d74ec7f29055a6684fa06fb3c2f6821c87dd38f9a458dfd2e8a1db28189" OR
   NOT SQLITE_H_HASH STREQUAL "919e7f2e8ed1d8f56ac17b412b8971c76aa5d1a879752cc6058f75e7d5910e1d")
  message(FATAL_ERROR "Pinned SQLite source/header changed; refusing to build")
endif()
add_library(marksix_sqlite STATIC "${SQLITE_DIR}/sqlite3.c")
target_include_directories(marksix_sqlite PUBLIC "${SQLITE_DIR}")
target_compile_definitions(marksix_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
set(MARKSIX_HTTPS hkjc_https.cpp)
if(APPLE)
  set(CMAKE_CXX_STANDARD 20)
  include(CheckCXXSourceCompiles)
  check_cxx_source_compiles("#include <stop_token>
#include <thread>
int main() { std::stop_source source; std::jthread thread([]{}); return source.stop_requested(); }" MARKSIX_CXX20_CANCELLATION)
  if(NOT MARKSIX_CXX20_CANCELLATION)
    message(FATAL_ERROR "C++20 stop_token/jthread support is required. Select Xcode 26 or newer; experimental libc++ is not supported.")
  endif()
  find_package(Qt6 6.10.2 EXACT REQUIRED COMPONENTS Core Network)
  set(MARKSIX_HTTPS hkjc_https_qt.cpp)
endif()
add_library(marksix_statistics STATIC platform.cpp hkjc_policy.cpp domain.cpp analysis.cpp forecast.cpp adaptive.cpp forecast_lab.cpp forecast_lab_replay.cpp forecast_lab_evaluation.cpp forecast_lab_controls.cpp forecast_lab_storage.cpp forecast_lab_ledger.cpp forecast_lab_control_worker.cpp adaptive_ledger.cpp adaptive_coordinator.cpp learning_progress.cpp archive.cpp import.cpp coverage.cpp presentation.cpp update_plan.cpp update_worker.cpp hkjc_provider.cpp ${MARKSIX_HTTPS})
target_compile_features(marksix_statistics PUBLIC cxx_std_20)
target_include_directories(marksix_statistics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(marksix_statistics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(marksix_statistics PRIVATE /W4 /WX /permissive- /utf-8 /fp:precise)
  target_link_libraries(marksix_statistics PRIVATE bcrypt winhttp)
else()
  target_compile_options(marksix_statistics PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -ffp-contract=off)
  target_link_libraries(marksix_statistics PRIVATE Qt6::Core Qt6::Network)
endif()
target_link_libraries(marksix_statistics PRIVATE marksix_sqlite)
add_executable(MarkSixStatisticsTests tests.cpp platform_tests.cpp adaptive_tests.cpp forecast_lab_tests.cpp adaptive_ledger_tests.cpp learning_progress_tests.cpp archive_tests.cpp reference_report.cpp update_plan_tests.cpp update_worker_tests.cpp hkjc_provider_tests.cpp)
target_link_libraries(MarkSixStatisticsTests PRIVATE marksix_statistics marksix_sqlite)
if(MSVC)
  target_compile_options(MarkSixStatisticsTests PRIVATE /W4 /WX /permissive- /utf-8 /fp:precise)
else()
  target_compile_options(MarkSixStatisticsTests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -ffp-contract=off)
endif()
set_target_properties(MarkSixStatisticsTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
add_test(NAME statistics_domain_math_forecast COMMAND MarkSixStatisticsTests)
set_tests_properties(statistics_domain_math_forecast PROPERTIES LABELS "statistics;non-loaded")
