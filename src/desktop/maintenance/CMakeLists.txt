cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_policy(SET CMP0091 NEW)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  project(MarkSixMaintenance LANGUAGES C CXX)
  enable_testing()
endif()

# This library deliberately has no Qt dependency: the uninstall executor must
# continue after the application and all of its Qt libraries have been removed.
set(SQLITE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../deps/sqlite-amalgamation-3530400")
if(NOT TARGET marksix_sqlite)
  file(SHA256 "${SQLITE_DIR}/sqlite3.c" SQLITE_C_HASH)
  file(SHA256 "${SQLITE_DIR}/sqlite3.h" SQLITE_H_HASH)
  if(NOT SQLITE_C_HASH STREQUAL "b1dd5d74ec7f29055a6684fa06fb3c2f6821c87dd38f9a458dfd2e8a1db28189" OR
     NOT SQLITE_H_HASH STREQUAL "919e7f2e8ed1d8f56ac17b412b8971c76aa5d1a879752cc6058f75e7d5910e1d")
    message(FATAL_ERROR "Pinned SQLite source/header changed; refusing to build")
  endif()
  add_library(marksix_sqlite STATIC "${SQLITE_DIR}/sqlite3.c")
  target_include_directories(marksix_sqlite PUBLIC "${SQLITE_DIR}")
  target_compile_definitions(marksix_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
endif()
if(WIN32)
  set(MARKSIX_PROFILE_SOURCE profile_discovery_windows.cpp application_discovery_windows.cpp native_dialog_windows.cpp installer_launch_windows.cpp release_http_windows.cpp payload_authority_windows.cpp uninstall_native_windows.cpp)
else()
  enable_language(OBJCXX)
  set(MARKSIX_PROFILE_SOURCE profile_discovery_macos.cpp application_discovery_macos.cpp native_dialog_macos.mm installer_launch_macos.mm release_http_macos.mm payload_authority_macos.mm uninstall_native_macos.cpp)
endif()
set(MARKSIX_MAINTENANCE_SOURCES uninstall_execution.cpp storage_inventory.cpp file_identity.cpp ownership_manifest.cpp profile_storage.cpp application_identity.cpp payload_inventory.cpp preserved_legacy.cpp native_runtime.cpp durable_files.cpp reset_journal.cpp private_channel.cpp user_cleanup.cpp user_cleanup_lifecycle.cpp installer_lease.cpp download_inventory.cpp uninstall_inventory.cpp elevation_request.cpp verified_removal.cpp workspace_registration.cpp machine_gate.cpp payload_authority.cpp uninstall_journal.cpp uninstall_metadata.cpp native_command.cpp native_report.cpp ${MARKSIX_PROFILE_SOURCE})
add_library(marksix_maintenance STATIC ${MARKSIX_MAINTENANCE_SOURCES})
target_compile_features(marksix_maintenance PUBLIC cxx_std_20)
set_target_properties(marksix_maintenance PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_include_directories(marksix_maintenance PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(marksix_maintenance PRIVATE marksix_sqlite)
if(MSVC)
  target_compile_options(marksix_maintenance PRIVATE /W4 /WX /permissive- /utf-8)
  target_link_libraries(marksix_maintenance PRIVATE bcrypt advapi32 shell32 ole32 uuid version user32 gdi32 winhttp)
else()
  target_compile_options(marksix_maintenance PRIVATE -Wall -Wextra -Werror)
  target_compile_options(marksix_maintenance PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(marksix_maintenance PRIVATE "-framework CoreFoundation" "-framework CoreServices" "-framework AppKit" "-framework Foundation" "-framework Security")
endif()
if(MSVC)
  # The Qt application uses /MD. Never mix its library objects into the helper:
  # build a separate native core and SQLite with /MT for standalone execution.
  add_library(marksix_helper_sqlite STATIC "${SQLITE_DIR}/sqlite3.c")
  target_include_directories(marksix_helper_sqlite PUBLIC "${SQLITE_DIR}")
  target_compile_definitions(marksix_helper_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
  add_library(marksix_helper_core STATIC ${MARKSIX_MAINTENANCE_SOURCES})
  target_compile_features(marksix_helper_core PUBLIC cxx_std_20)
  target_include_directories(marksix_helper_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(marksix_helper_core PRIVATE marksix_helper_sqlite bcrypt advapi32 shell32 ole32 uuid version user32 gdi32 winhttp)
  target_compile_options(marksix_helper_core PRIVATE /W4 /WX /permissive- /utf-8)
  set_target_properties(marksix_helper_sqlite marksix_helper_core PROPERTIES
    MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
  set(MARKSIX_HELPER_CORE marksix_helper_core)
else()
  set(MARKSIX_HELPER_CORE marksix_maintenance)
endif()
if(WIN32)
  set(MARKSIX_FULL_PLATFORM full_uninstall_windows.cpp)
else()
  set(MARKSIX_FULL_PLATFORM full_uninstall_macos.cpp)
endif()
add_executable(MarkSixMaintenance helper_main.cpp helper_lifecycle.cpp full_uninstall.cpp ${MARKSIX_FULL_PLATFORM})
target_link_libraries(MarkSixMaintenance PRIVATE ${MARKSIX_HELPER_CORE})
target_compile_features(MarkSixMaintenance PRIVATE cxx_std_20)
set_target_properties(MarkSixMaintenance PROPERTIES OUTPUT_NAME "mark-six-maintenance"
  MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(MSVC)
  target_compile_options(MarkSixMaintenance PRIVATE /W4 /WX /permissive- /utf-8)
  target_link_options(MarkSixMaintenance PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
else()
  target_compile_options(MarkSixMaintenance PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME storage_maintenance_native_runtime COMMAND MarkSixMaintenance --verify-runtime)
add_executable(MarkSixUninstallFixture tests/uninstall_native_fixture.cpp helper_lifecycle.cpp full_uninstall.cpp ${MARKSIX_FULL_PLATFORM})
target_link_libraries(MarkSixUninstallFixture PRIVATE ${MARKSIX_HELPER_CORE})
target_compile_features(MarkSixUninstallFixture PRIVATE cxx_std_20)
set_target_properties(MarkSixUninstallFixture PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(MSVC)
  target_compile_options(MarkSixUninstallFixture PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(MarkSixUninstallFixture PRIVATE -Wall -Wextra -Werror)
endif()
set_tests_properties(storage_maintenance_native_runtime PROPERTIES LABELS "maintenance;native" TIMEOUT 15)
if(WIN32)
  add_executable(MarkSixMaintenanceLifecycleFixture tests/windows_finalizer_fixture.cpp helper_lifecycle.cpp)
  target_link_libraries(MarkSixMaintenanceLifecycleFixture PRIVATE ${MARKSIX_HELPER_CORE})
  target_compile_options(MarkSixMaintenanceLifecycleFixture PRIVATE /utf-8 /W4 /permissive-)
  set_target_properties(MarkSixMaintenanceLifecycleFixture PROPERTIES
    MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
  add_test(NAME storage_maintenance_windows_finalizer COMMAND MarkSixMaintenanceLifecycleFixture)
  set_tests_properties(storage_maintenance_windows_finalizer PROPERTIES LABELS "maintenance;native;disposable" TIMEOUT 180 SKIP_RETURN_CODE 77)
endif()
add_executable(MarkSixMaintenanceTests tests/storage_inventory_tests.cpp helper_lifecycle.cpp full_uninstall.cpp ${MARKSIX_FULL_PLATFORM})
if(WIN32)
  enable_language(RC)
  target_sources(MarkSixMaintenanceTests PRIVATE tests/product_fixture.rc)
  target_compile_options(MarkSixMaintenanceTests PRIVATE /utf-8 /W4 /permissive-)
endif()
target_link_libraries(MarkSixMaintenanceTests PRIVATE marksix_maintenance marksix_sqlite)
add_test(NAME storage_maintenance_inventory COMMAND MarkSixMaintenanceTests)
set_tests_properties(storage_maintenance_inventory PROPERTIES LABELS "maintenance;native" TIMEOUT 60)
